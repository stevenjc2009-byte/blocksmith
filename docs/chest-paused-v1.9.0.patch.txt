def tile_chest_top(rng):
    """The chest's lid, v1.9.0 "Storage". Reuses tile_planks' board palette but draws
    VERTICAL boards (planks draws horizontal) plus a horizontal metal strap and a centered
    dark latch square, so the chest reads as its own object rather than as a re-tinted
    plank block. Fully opaque — no alpha-0 texels — because this is a real block face
    (world/registry.c's chest row points FACE_TOP directly at it), unlike the item icons
    at slots 48..56 which DO carry alpha and must never be reused for a block face.
    """
    plank_base = (168, 132, 88)
    plank_warm = (182, 146, 100)
    plank_cool = (150, 116, 78)
    seam       = (96, 70, 46)
    strap      = (132, 130, 124)
    strap_dark = (86, 84, 80)
    latch      = (54, 52, 48)
    img = Image.new("RGB", (TILE_PX, TILE_PX))
    px = img.load()
    board_w = 4
    for x in range(TILE_PX):
        board = x // board_w
        tint = blend(plank_cool, plank_warm, (board * 0.31 + 0.15) % 1.0)
        for y in range(TILE_PX):
            px[x, y] = tint if rng.random() < 0.55 else blend(tint, plank_base, 0.5)
    # (board seams every board_w px, one horizontal strap band, one centered latch square —
    #  drawn on top of the base fill above; exact pixel rows were tuned by eye against a
    #  256x256 zoom render and are not reproduced here — re-render and re-look, per this
    #  project's own "render at zoom and LOOK at it" rule, rather than trusting this comment)
    return img
